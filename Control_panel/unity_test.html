Unity C# Integration Code

// RobotArmReceiver.cs - 掛在 Unity 的 GameObject 上 using UnityEngine; using System; using NativeWebSocket; // 安裝: https://github.com/endel/NativeWebSocket public class RobotArmReceiver : MonoBehaviour { public string serverUrl = "ws://192.168.x.x:9090/unity"; private WebSocket ws; // θ1~θ5 (radians) public double[] theta = new double[5]; [Serializable] private class ThetaData { public double theta1, theta2, theta3, theta4, theta5; } async void Start() { ws = new WebSocket(serverUrl); ws.OnOpen += () => Debug.Log("Connected to bridge!"); ws.OnClose += (code) => Debug.Log("Disconnected."); ws.OnMessage += (bytes) => { var msg = System.Text.Encoding.UTF8.GetString(bytes); var data = JsonUtility.FromJson<ThetaData>(msg); theta[0] = data.theta1; theta[1] = data.theta2; theta[2] = data.theta3; theta[3] = data.theta4; theta[4] = data.theta5; }; await ws.Connect(); } void Update() { if (ws != null) ws.DispatchMessageQueue(); // 用 theta[0]~theta[4] 來驅動你的 Joint Transform } async void OnApplicationQuit() { if (ws != null) await ws.Close(); } }
